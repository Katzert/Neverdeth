1721: // Ajusta el refresco REAL del monitor (Hz) al limite de FPS elegido. El combo
1722: // del menu promete "60 Hz" pero solo se limitaba el present: el monitor seguia
1723: // a su frecuencia nativa. Se usa CDS_FULLSCREEN (transitorio), asi que al salir
1724: // del juego Windows restaura el modo original.
1725: static void NdApplyDisplayRefresh(int hz) {
1726:   if (hz <= 0) {
1727:     ChangeDisplaySettingsW(nullptr, 0);  // restaurar el modo por defecto
1728:     REXLOG_INFO("neverdead: [CONFIG-UI] Refresco del monitor restaurado al modo por defecto");
1729:     return;
1730:   }
1731:   DEVMODEW cur{};
1732:   cur.dmSize = sizeof(cur);
1733:   if (!EnumDisplaySettingsW(nullptr, ENUM_CURRENT_SETTINGS, &cur)) {
1734:     return;
1735:   }
1736:   DEVMODEW best{};
1737:   bool found = false;
1738:   int best_diff = 1 << 30;
1739:   for (DWORD i = 0;; ++i) {
1740:     DEVMODEW dm{};
1741:     dm.dmSize = sizeof(dm);
1742:     if (!EnumDisplaySettingsW(nullptr, i, &dm)) {
1743:       break;
1744:     }
1745:     if (dm.dmPelsWidth != cur.dmPelsWidth || dm.dmPelsHeight != cur.dmPelsHeight ||
1746:         dm.dmBitsPerPel != cur.dmBitsPerPel) {
1747:       continue;
1748:     }
1749:     const int freq = static_cast<int>(dm.dmDisplayFrequency);
1750:     const int diff = freq > hz ? freq - hz : hz - freq;
1751:     if (diff < best_diff) {
1752:       best_diff = diff;
1753:       best = dm;
1754:       found = true;
1755:     }
1756:   }
1757:   if (!found || best_diff > 3) {
1758:     REXLOG_INFO("neverdead: [CONFIG-UI] no hay modo de {} Hz a {}x{}; se mantiene {} Hz", hz,
1759:                 static_cast<int>(cur.dmPelsWidth), static_cast<int>(cur.dmPelsHeight),
1760:                 static_cast<int>(cur.dmDisplayFrequency));
1761:     return;
1762:   }
1763:   if (best.dmDisplayFrequency == cur.dmDisplayFrequency) {
1764:     REXLOG_INFO("neverdead: [CONFIG-UI] el monitor ya esta a {} Hz",
1765:                 static_cast<int>(cur.dmDisplayFrequency));
1766:     return;
1767:   }
1768:   const LONG res = ChangeDisplaySettingsW(&best, CDS_FULLSCREEN);
1769:   REXLOG_INFO("neverdead: [CONFIG-UI] Refresco del monitor: {} Hz -> {} Hz (resultado {})",
1770:               static_cast<int>(cur.dmDisplayFrequency),
1771:               static_cast<int>(best.dmDisplayFrequency), static_cast<int>(res));
1772: }
